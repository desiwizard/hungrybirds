function playRubberStretch(pressureRatio) {
if (!sfxEnabled || !isAiming || gameState !== 'PLAY') return;
stretchAudio.volume = Math.min(0.65, 0.2 + pressureRatio * 0.45);
stretchAudio.playbackRate = 0.8 + pressureRatio * 0.8;
if (!isStretchPlaying) {
isStretchPlaying = true;
stretchAudio.currentTime = 0;
stretchAudio.play().catch(() => {});
}
}
function stopRubberStretch() {
isStretchPlaying = false;
try {
stretchAudio.pause();
stretchAudio.currentTime = 0;
} catch(e) {}
}

}

function getCoords(e) {
const rect = canvas.getBoundingClientRect();
const cx = e.touches ? e.touches[0].clientX : e.clientX;
const cy = e.touches ? e.touches[0].clientY : e.clientY;
return {
x: (cx - rect.left) * (canvas.width / rect.width),
y: (cy - rect.top) * (canvas.height / rect.height)
};
}
function onDown(pos) {
initAudioSystem();
if (gameState !== 'PLAY' || bird.inFlight || birdsLeft <= 0 || isWinningTransition) return;
const d = Math.hypot(pos.x - SLING_X, pos.y - SLING_Y);
if (d < 75) {
isAiming = true;
}
}
function onMove(pos) {
if (!isAiming || gameState !== 'PLAY' || bird.inFlight || isWinningTransition) return;
const dx = pos.x - SLING_X;
const dy = pos.y - SLING_Y;
const dist = Math.hypot(dx, dy);
if (dist > MAX_DRAG) {
dragPos.x = SLING_X + (dx / dist) * MAX_DRAG;
dragPos.y = SLING_Y + (dy / dist) * MAX_DRAG;
} else {
dragPos.x = pos.x;
dragPos.y = pos.y;
}
const pressureRatio = Math.min(1.0, Math.hypot(SLING_X - dragPos.x, SLING_Y - dragPos.y) / MAX_DRAG);
if (pressureRatio > 0.12) {
playRubberStretch(pressureRatio);
} else {
stopRubberStretch();
}
}
function onUp() {
if (!isAiming) return;
isAiming = false;
stopRubberStretch();
const pullX = SLING_X - dragPos.x;
const pullY = SLING_Y - dragPos.y;
const pullDist = Math.hypot(pullX, pullY);
if (pullDist > 12) {
const tier = getCurrentTier();
bird.inFlight = true;
hasFirstFired = true;
birdsLeft--;
updateHUD();
bird.bounces = 0;
bird.stoppedFrames = 0;
bird.hitBlocksCooldown = {};
bird.x = dragPos.x;
bird.y = dragPos.y;
bird.vx = pullX * tier.power;
bird.vy = pullY * tier.power;
playLaunchAudio();
} else {
dragPos = { x: SLING_X, y: SLING_Y };
}
}

function loop() {
if (gameState === 'PLAY') {
if (hasFirstFired && !isWinningTransition) {
const diff = DIFFICULTY_CONFIG[currentDifficulty];
energy -= 0.016 * diff.drainMult;
if (energy <= 0) {
energy = 0;
gameOver("Giant Bird Starved!", "Energy depleted! Keep the Giant Bird fed.");
}
}
if (bird.inFlight) {
bird.vy += 0.32;
bird.x += bird.vx;
bird.y += bird.vy;
if (bird.y + bird.r >= GROUND_Y) {
bird.y = GROUND_Y - bird.r;
if (Math.abs(bird.vy) > 1.5 && bird.bounces < 3) {
bird.vy = -Math.abs(bird.vy) * 0.55;
bird.vx *= 0.76;
bird.bounces++;
playSmashAudio();
} else {
bird.vy = 0;
bird.vx *= 0.88;
bird.stoppedFrames++;
if (bird.stoppedFrames > 40 || Math.abs(bird.vx) < 0.15) {
energy = Math.max(0, energy - 4);
onBirdFinishFlight();
}
}
}
for (let i = blocks.length - 1; i >= 0; i--) {
const b = blocks[i];
const closestX = Math.max(b.x, Math.min(bird.x, b.x + b.w));
const closestY = Math.max(b.y, Math.min(bird.y, b.y + b.h));
const distX = bird.x - closestX;
const distY = bird.y - closestY;
const distSq = (distX * distX) + (distY * distY);
if (distSq < (bird.r * bird.r)) {
if (!bird.hitBlocksCooldown[b.id]) {
bird.hitBlocksCooldown[b.id] = true;
const impactSpeed = Math.hypot(bird.vx, bird.vy);
const damage = Math.max(48, Math.min(65, Math.floor(impactSpeed * 3.8)));
b.hp -= damage;
score += 25;
playSmashAudio();
bird.vx = -bird.vx * 0.40;
bird.vy *= 0.6;
if (b.hp <= 0) {
blocks.splice(i, 1);
}
}
}
}
foods.forEach(f => {
if (f.alive && Math.hypot(bird.x - f.x, bird.y - f.y) < bird.r + f.r) {
f.alive = false;
score += f.pts;
foodFlights.push({
startX: f.x, startY: f.y, x: f.x, y: f.y, color: f.color, r: f.r, progress: 0
});
}
});
if (bird.x > 970 || bird.x < -30) {
energy = Math.max(0, energy - 4);
onBirdFinishFlight();
}
}
foods.forEach(f => {
if (f.alive && f.y + f.r < GROUND_Y) {
let sup = false;
for (let b of blocks) {
if (f.x > b.x && f.x < b.x + b.w && b.y >= f.y + f.r - 4 && b.y <= f.y + f.r + 10) {
sup = true; f.y = b.y - f.r; f.vy = 0; break;
}
}
if (!sup) {
f.vy = (f.vy || 0) + 0.35;
f.y += f.vy;
if (f.y + f.r >= GROUND_Y) { f.y = GROUND_Y - f.r; f.vy = 0; }
}
}
});
for (let i = foodFlights.length - 1; i >= 0; i--) {
const fl = foodFlights[i];
fl.progress += 0.05;
const targetX = BIG_BIRD_X + 26;
const targetY = BIG_BIRD_Y - 4;
fl.x += (targetX - fl.x) * 0.15;
fl.y += (targetY - fl.y) * 0.15;
if (fl.progress >= 0.95 || Math.hypot(fl.x - targetX, fl.y - targetY) < 18) {
energy = Math.min(100, energy + 28);
chompTimer = 25;
playBigBirdCrunch();
foodFlights.splice(i, 1);
checkAllFoodFinished();
}
}
if (chompTimer > 0) chompTimer--;
}
updateHUD();
drawScene();
requestAnimationFrame(loop);
}
